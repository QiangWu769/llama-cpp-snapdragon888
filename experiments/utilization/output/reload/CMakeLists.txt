cmake_minimum_required(VERSION 3.16)
project(hmx_output_reload LANGUAGES CXX)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Matching isolated output-source directory")
set(LLAMA_BUILD_DIR "" CACHE PATH "Existing matching isolated output-build directory")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h" OR
   NOT EXISTS "${LLAMA_SOURCE_DIR}/ggml/include/ggml-htp-output-pack.h" OR
   NOT EXISTS "${LLAMA_BUILD_DIR}/src/libllama.so")
    message(FATAL_ERROR "Provide built matching LLAMA_SOURCE_DIR and LLAMA_BUILD_DIR")
endif()
add_library(existing_llama SHARED IMPORTED)
set_target_properties(existing_llama PROPERTIES IMPORTED_LOCATION "${LLAMA_BUILD_DIR}/src/libllama.so")
set(base_library "${LLAMA_BUILD_DIR}/ggml/src/libggml-base.so")
if(NOT EXISTS "${base_library}")
    message(FATAL_ERROR "Matching libggml-base.so missing")
endif()
add_library(existing_ggml_base SHARED IMPORTED)
set_target_properties(existing_ggml_base PROPERTIES IMPORTED_LOCATION "${base_library}")
add_executable(hmx-output-reload reload.cpp)
target_compile_features(hmx-output-reload PRIVATE cxx_std_17)
target_compile_options(hmx-output-reload PRIVATE -O2 -Wall -Wextra -Werror)
target_include_directories(hmx-output-reload PRIVATE "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include")
target_link_libraries(hmx-output-reload PRIVATE existing_llama existing_ggml_base)
foreach(directory IN ITEMS "${LLAMA_BUILD_DIR}/ggml/src" "${LLAMA_BUILD_DIR}/ggml/src/ggml-htp" "${LLAMA_BUILD_DIR}/ggml/src/ggml-cpu")
    target_link_options(hmx-output-reload PRIVATE "-Wl,-rpath-link,${directory}")
endforeach()
