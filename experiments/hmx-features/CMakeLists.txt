cmake_minimum_required(VERSION 3.14)
project(hmx_probe C)
set(CMAKE_C_STANDARD 11)
include(${HEXAGON_SDK_ROOT}/build/cmake/hexagon_fun.cmake)
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}
    ${HEXAGON_SDK_ROOT}/incs ${HEXAGON_SDK_ROOT}/incs/stddef
    ${HEXAGON_SDK_ROOT}/rtos/qurt)
if(OS_TYPE MATCHES "HLOS")
    add_library(hmx_probe SHARED ${CMAKE_CURRENT_BINARY_DIR}/hmx_probe_stub.c)
    build_idl(hmx_probe.idl hmx_probe)
    choose_dsprpc(${DSP_TYPE} dsprpc)
    link_custom_library(hmx_probe ${dsprpc})
    add_executable(hmx_probe_test host.c)
    set_common_compile_and_link_options(hmx_probe_test)
    target_link_libraries(hmx_probe_test hmx_probe m)
    copy_binaries(hmx_probe hmx_probe_test)
else()
    add_library(hmx_probe_skel SHARED ${CMAKE_CURRENT_BINARY_DIR}/hmx_probe_skel.c dsp.c)
    build_idl(hmx_probe.idl hmx_probe_skel)
    target_compile_options(hmx_probe_skel PRIVATE -mhmx)
    copy_binaries(hmx_probe_skel)
endif()
